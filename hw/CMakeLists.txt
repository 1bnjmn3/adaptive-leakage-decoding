cmake_minimum_required(VERSION 3.16)
project(uf_decode CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_FLAGS_RELEASE "-O2")
add_library(uf STATIC src/uf.cpp)
target_include_directories(uf PUBLIC src)
add_executable(uf_decode src/main.cpp)
target_link_libraries(uf_decode uf)
add_executable(test_uf tests/test_uf.cpp)
target_link_libraries(test_uf uf)
enable_testing()
add_test(NAME test_uf COMMAND test_uf)
